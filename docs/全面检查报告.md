```python
def _create_backup(self):
    if self.config_file.exists():
        timestamp = datetime.now().strftime('%Y%m%d_%H%M%S')
        backup_file = self.backup_dir / f'channel_configs_{timestamp}.json'
        shutil.copy2(self.config_file, backup_file)
        self._cleanup_old_backups()

def _cleanup_old_backups(self):
    backup_files = sorted(self.backup_dir.glob('channel_configs_*.json'), reverse=True)
    for old_backup in backup_files[10:]:
        old_backup.unlink()
```

def export_config(self):

        self,
        "导出外部告警渠道配置",
        "",
        "JSON文件 (*.json);;所有文件 (*.*)"
    )

    if file_path:
        if self.config_persistence:
            success = self.config_persistence.export_config(file_path)
            if success:
