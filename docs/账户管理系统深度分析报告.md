## 🚨 二、发现的问题

### 2.1 严重问题（P0）

#### 问题1: 数据持久化缺失
**严重程度**: 🔴 **严重**
**影响范围**: 所有账户数据

**问题描述**:
- AccountManager使用纯内存存储
- 程序重启后所有账户数据丢失
- 无法保存账户配置和历史记录

**影响**:
- 用户无法持久化账户信息
- 每次重启需要重新创建账户
- 无法查询历史账户数据
- 无法进行账户数据分析

**示例**:
```python
# 当前实现
```

### 3.1 数据持久化解决方案

#### 方案1: 创建AccountRepository

**文件**: `core/trading/account_repository.py`

```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
账户仓储

负责账户、持仓、资金数据的持久化
"""

from loguru import logger
from datetime import datetime
from typing import List, Optional, Dict, Any
from uuid import uuid4

from core.trading.account_models import (
    Account, Position, FundInfo, AccountQuery, PositionQuery,
    AccountStatus, PositionSide
)
from core.containers import ServiceContainer
from core.events import EventBus


class AccountRepository:
    """账户仓储"""
    
    def __init__(self, db_service: DatabaseService, event_bus: EventBus):
        """
        初始化账户仓储
        
        Args:
            db_service: 数据库服务
            event_bus: 事件总线
        """
        self.db_service = db_service
        self.event_bus = event_bus
        
        # 初始化数据库表
        self._initialize_tables()
        
        logger.info("账户仓储初始化完成")
    
    def _initialize_tables(self):
        """初始化数据库表"""
        try:
            # 创建账户表
            self.db_service.execute("""
                CREATE TABLE IF NOT EXISTS accounts (
                    account_id TEXT PRIMARY KEY,
                    account_name TEXT NOT NULL,
                    account_type TEXT NOT NULL,
                    status TEXT NOT NULL,
                    balance REAL NOT NULL,
                    available_balance REAL NOT NULL,
                    frozen_balance REAL NOT NULL,
                    market_value REAL NOT NULL,
                    total_assets REAL NOT NULL,
                    profit_loss REAL NOT NULL,
                    profit_loss_ratio REAL NOT NULL,
                    create_time TEXT NOT NULL,
                    update_time TEXT NOT NULL,
                    user_id TEXT DEFAULT 'system',
                    trading_day TEXT DEFAULT '',
                    risk_level TEXT DEFAULT 'normal',
                    margin_ratio REAL DEFAULT 0.0,
                    maintenance_margin REAL DEFAULT 0.0,
                    metadata TEXT DEFAULT '{}'
                )
            """)
            
            # 创建持仓表
            self.db_service.execute("""
                CREATE TABLE IF NOT EXISTS positions (
                    position_id TEXT PRIMARY KEY,
                    account_id TEXT NOT NULL,
                    asset_type TEXT NOT NULL,
                    stock_code TEXT NOT NULL,
                    stock_name TEXT NOT NULL,
                    side TEXT NOT NULL,
                    quantity INTEGER NOT NULL,
                    available_quantity INTEGER NOT NULL,
                    open_price REAL NOT NULL,
                    current_price REAL NOT NULL,
                    market_value REAL NOT NULL,
                    cost_price REAL NOT NULL,
                    cost_value REAL NOT NULL,
                    profit_loss REAL NOT NULL,
                    profit_loss_ratio REAL NOT NULL,
                    open_time TEXT NOT NULL,
                    update_time TEXT NOT NULL,
                    commission REAL DEFAULT 0.0,
                    metadata TEXT DEFAULT '{}',
                    FOREIGN KEY (account_id) REFERENCES accounts(account_id)
                )
            """)
            
            # 创建资金信息表
            self.db_service.execute("""
                CREATE TABLE IF NOT EXISTS fund_infos (
                    account_id TEXT PRIMARY KEY,
                    total_balance REAL NOT NULL,
                    available_balance REAL NOT NULL,
                    frozen_balance REAL NOT NULL,
                    market_value REAL NOT NULL,
                    total_assets REAL NOT NULL,
                    profit_loss REAL NOT NULL,
                    profit_loss_ratio REAL NOT NULL,
                    margin_used REAL NOT NULL,
                    margin_available REAL NOT NULL,
                    maintenance_margin REAL NOT NULL,
                    update_time TEXT NOT NULL,
                    metadata TEXT DEFAULT '{}',
                    FOREIGN KEY (account_id) REFERENCES accounts(account_id)
                )
            """)
            
            logger.info("账户数据库表初始化完成")
            
        except Exception as e:
            logger.error(f"初始化账户数据库表失败: {e}")
            raise
    
    def create_account(self, account: Account) -> bool:
        """创建账户"""
        try:
            self.db_service.execute("""
                INSERT INTO accounts (
                    account_id, account_name, account_type, status,
                    balance, available_balance, frozen_balance, market_value,
                    total_assets, profit_loss, profit_loss_ratio,
                    create_time, update_time, user_id, trading_day,
                    risk_level, margin_ratio, maintenance_margin, metadata
                ) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
            """, (
                account.account_id, account.account_name, account.account_type,
                account.status.value, account.balance, account.available_balance,
                account.frozen_balance, account.market_value, account.total_assets,
                account.profit_loss, account.profit_loss_ratio,
                account.create_time.isoformat(), account.update_time.isoformat(),
                account.user_id, account.trading_day, account.risk_level,
                account.margin_ratio, account.maintenance_margin,
                str(account.metadata)
            ))
            
            logger.info(f"账户创建成功: {account.account_id}")
            return True
            
        except Exception as e:
            logger.error(f"创建账户失败: {e}")
            return False
    
    def update_account(self, account: Account) -> bool:
        """更新账户"""
        try:
            self.db_service.execute("""
                UPDATE accounts SET
                    account_name = ?, account_type = ?, status = ?,
                    balance = ?, available_balance = ?, frozen_balance = ?,
                    market_value = ?, total_assets = ?, profit_loss = ?,
                    profit_loss_ratio = ?, update_time = ?, user_id = ?,
                    trading_day = ?, risk_level = ?, margin_ratio = ?,
                    maintenance_margin = ?, metadata = ?
                WHERE account_id = ?
            """, (
                account.account_name, account.account_type, account.status.value,
                account.balance, account.available_balance, account.frozen_balance,
                account.market_value, account.total_assets, account.profit_loss,
                account.profit_loss_ratio, account.update_time.isoformat(),
                account.user_id, account.trading_day, account.risk_level,

    def get_account(self, account_id: str) -> Optional[Account]:
        """获取账户"""
        try:

            return None
    
    def query_accounts(self, query: AccountQuery) -> List[Account]:
        """查询账户列表"""
        try:
            sql = "SELECT * FROM accounts WHERE 1=1"
            params = []
            
            if query.account_id:
                sql += " AND account_id = ?"
                params.append(query.account_id)
            
            if query.user_id:
                sql += " AND user_id = ?"
                params.append(query.user_id)
            
            if query.account_type:
                sql += " AND account_type = ?"
                params.append(query.account_type)
            
            if query.status:
                sql += " AND status = ?"
                params.append(query.status.value)
            
            # 排序
            if query.sort_by == "create_time":
                sql += f" ORDER BY create_time {query.sort_order}"
            elif query.sort_by == "update_time":
                sql += f" ORDER BY update_time {query.sort_order}"
            elif query.sort_by == "balance":
                sql += f" ORDER BY balance {query.sort_order}"
            
            # 分页
            if query.limit:
                sql += " LIMIT ? OFFSET ?"
                params.append(query.limit)
                params.append(query.offset)
            
            rows = self.db_service.fetch_all(sql, tuple(params))
            
            accounts = []
            for row in rows:
                accounts.append(Account(
                    account_id=row[0],
                    account_name=row[1],
                    account_type=row[2],
                    status=AccountStatus(row[3]),
                    balance=row[4],
                    available_balance=row[5],
                    frozen_balance=row[6],
                    market_value=row[7],
                    total_assets=row[8],
                    profit_loss=row[9],
                    profit_loss_ratio=row[10],
                    create_time=datetime.fromisoformat(row[11]),
                    update_time=datetime.fromisoformat(row[12]),
                    user_id=row[13],
                    trading_day=row[14],
                    risk_level=row[15],
                    margin_ratio=row[16],
                    maintenance_margin=row[17],
                    metadata=eval(row[18]) if row[18] else {}
                ))
            
            return accounts
            
        except Exception as e:
            logger.error(f"查询账户列表失败: {e}")
            return []

                open_time=datetime.fromisoformat(row[15]),
                update_time=datetime.fromisoformat(row[16]),

            return None
    
    def query_positions(self, query: PositionQuery) -> List[Position]:
        """查询持仓列表"""
        try:
            sql = "SELECT * FROM positions WHERE 1=1"
            params = []
            
            if query.account_id:
                sql += " AND account_id = ?"
                params.append(query.account_id)
            
            if query.asset_type:
                sql += " AND asset_type = ?"
                params.append(query.asset_type.value)
            
            if query.stock_code:
                sql += " AND stock_code = ?"
                params.append(query.stock_code)
            
            if query.side:
                sql += " AND side = ?"
                params.append(query.side.value)
            
            # 排序
            if query.sort_by == "open_time":
                sql += f" ORDER BY open_time {query.sort_order}"
            elif query.sort_by == "update_time":
                sql += f" ORDER BY update_time {query.sort_order}"
            elif query.sort_by == "market_value":
                sql += f" ORDER BY market_value {query.sort_order}"
            
            # 分页
            if query.limit:
                sql += " LIMIT ? OFFSET ?"
                params.append(query.limit)
                params.append(query.offset)
            
            rows = self.db_service.fetch_all(sql, tuple(params))
            
            positions = []
            for row in rows:
                positions.append(Position(
                    position_id=row[0],
                    account_id=row[1],
                    asset_type=AssetType(row[2]),
                    stock_code=row[3],
                    stock_name=row[4],
                    side=PositionSide(row[5]),
                    quantity=row[6],
                    available_quantity=row[7],
                    open_price=row[8],

                    profit_loss_ratio=row[14],
                    open_time=datetime.fromisoformat(row[15]),
                    update_time=datetime.fromisoformat(row[16]),
                    commission=row[17],
                    metadata=eval(row[18]) if row[18] else {}
                ))
            
            return positions
            
        except Exception as e:
            logger.error(f"查询持仓列表失败: {e}")
            return []
    
    def delete_position(self, position_id: str) -> bool:
```
